const Backdrop = () => {

      phaseOffset: (i * Math.PI) / 4.5,
    }));
  }, []);

  return (
    <div
      style={{
        position: "absolute",
        left: 0,
        top: 0,
        width: FRAME.w,
        height: FRAME.h,
        zIndex: 0,
        pointerEvents: "none",
        backgroundColor: PALETTE.base,
        overflow: "hidden",
      }}
    >
      <svg style={SVG_FRAME}>
        <defs>
          <radialGradient id="hdr-em-core" cx="50%" cy="50%" r="50%">
            <stop offset="0%" stopColor="#00E5FF" stopOpacity="0.12" />
            <stop offset="50%" stopColor="#E040FB" stopOpacity="0.06" />
            <stop offset="100%" stopColor="#080B14" stopOpacity="0" />
          </radialGradient>
          <linearGradient id="hdr-wave-electric" x1="0%" y1="0%" x2="100%" y2="0%">
            <stop offset="0%" stopColor="#00E5FF" stopOpacity="0.0" />
            <stop offset="25%" stopColor="#00E5FF" stopOpacity="0.18" />
            <stop offset="75%" stopColor="#00E5FF" stopOpacity="0.18" />
            <stop offset="100%" stopColor="#00E5FF" stopOpacity="0.0" />

          <linearGradient id="hdr-wave-magnetic" x1="0%" y1="0%" x2="100%" y2="0%">
            <stop offset="0%" stopColor="#E040FB" stopOpacity="0.0" />
            <stop offset="25%" stopColor="#E040FB" stopOpacity="0.15" />
            <stop offset="75%" stopColor="#E040FB" stopOpacity="0.15" />
            <stop offset="100%" stopColor="#E040FB" stopOpacity="0.0" />
          </linearGradient>
        </defs>

        <rect
          x={ZONES.stage.x}
          y={ZONES.stage.y}
          width={ZONES.stage.w}
          height={ZONES.stage.h}
          fill="url(#hdr-em-core)"
          style={{ opacity: 0.9 }}
        />

        {fieldLines.map((line) => {
          const isElectric = line.id % 2 === 0;
          const stroke = isElectric ? "url(#hdr-wave-electric)" : "url(#hdr-wave-magnetic)";
          const drift = Math.sin(time * 0.9 + line.phaseOffset) * 12;
          const segments = 16;
          const step = FRAME.w / segments;

          let pathData = `M 0 ${line.yBase + drift + Math.sin(time * 1.5 + line.phaseOffset) * line.amplitude}`;
          for (let s = 1; s <= segments; s++) {
            const x = s * step;
            const wave = Math.sin(x * line.frequency + time * 1.8 + line.phaseOffset) * line.amplitude;
            const y = line.yBase + drift + wave;
            pathData += ` L ${x.toFixed(1)} ${y.toFixed(1)}`;
          }

          return (
            <path
              key={line.id}
              d={pathData}
              fill="none"
              stroke={stroke}
              strokeWidth={1.5}
              strokeDasharray={isElectric ? "8 6" : "4 8"}
              strokeOpacity={0.8}
            />
          );
        })}
      </svg>
    </div>
  );
};

const Arrow = ({ from, to, color = PALETTE.accent, width = 3 }) => {
  const { x: x1, y: y1 } = from;
  const { x: x2, y: y2 } = to;
  const dx = x2 - x1;
  const dy = y2 - y1;
  const angle = Math.atan2(dy, dx);
  const headLen = Math.max(12, width * 3.5);
  const headAngle = Math.PI / 6;

  const leftX = x2 - headLen * Math.cos(angle - headAngle);
  const leftY = y2 - headLen * Math.sin(angle - headAngle);
  const rightX = x2 - headLen * Math.cos(angle + headAngle);
  const rightY = y2 - headLen * Math.sin(angle + headAngle);

  return (
    <svg style={SVG_FRAME}>
      <line
        x1={x1}
        y1={y1}
        x2={x2}
        y2={y2}
        stroke={color}
        strokeWidth={width}
        strokeLinecap="round"
      />
      <polygon
        points={`${x2},${y2} ${leftX},${leftY} ${rightX},${rightY}`}
        fill={color}
      />
    </svg>
  );
};

const AngleArc = ({ vertex, angle1, angle2, radius = 45, color = PALETTE.accent }) => {
  const { x, y } = vertex;
  const a1 = Math.min(angle1, angle2);
  const a2 = Math.max(angle1, angle2);
  const diff = a2 - a1;
  const largeArcFlag = diff > Math.PI ? 1 : 0;

  const startX = x + radius * Math.cos(a1);
  const startY = y + radius * Math.sin(a1);
  const endX = x + radius * Math.cos(a2);
  const endY = y + radius * Math.sin(a2);

  const pathData = `M ${startX} ${startY} A ${radius} ${radius} 0 ${largeArcFlag} 1 ${endX} ${endY}`;

  return (
    <svg style={SVG_FRAME}>
      <path
        d={pathData}
        fill="none"
        stroke={color}
        strokeWidth={2}
        strokeDasharray="4 3"
      />
    </svg>
  );
};

const S4_GRID_COLS = 12;
const S4_FIELD_RECT = { x: 420, y: 320, w: 1080, h: 480 };

function Scene4() {
  const frame = useCurrentFrame();

  const titleProgress = spring({
    frame,
    fps: 30,
    config: { damping: 14, stiffness: 90 },
  });

  const gridAlpha = interpolate(frame, [15, 45], [0, 0.85], CLAMP);

  // Trajectory mechanics:
  // Phase 1: Frame 0..60 -> straight entry from left (x: 160 -> 420, y: 640)
  // Phase 2: Frame 60..480 -> circular motion in magnetic field (center: 420, 360, radius: 280)
  //          starts at angle -PI/2 (bottom: 420, 640) and curves upward/rightward
  // Phase 3: Frame 480..566 -> continuous loop motion / exit stabilization

  const r = 260;
  const centerX = 440;
  const centerY = 400;

  // Angular progress through magnetic field
  const theta = interpolate(frame, [50, 480],
    [-Math.PI / 2, Math.PI / 2 + 0.35],
    CLAMP
  );

  let chargeX = 180;
  let chargeY = 660;
  let velAngle = 0; // horizontal right

  if (frame < 50) {
    chargeX = interpolate(frame, [0, 50], [180, centerX], CLAMP);
    chargeY = 660;
    velAngle = 0;
  } else {
    chargeX = centerX + r * Math.cos(theta);
    chargeY = centerY - r * Math.sin(theta);
    // Velocity is tangent to path (perpendicular to radius vector pointing inward)
    // Position vector from center: (cos(theta), -sin(theta))
    // Tangent vector: (-sin(theta), -cos(theta)) in screen coords
    velAngle = Math.atan2(-Math.cos(theta), -Math.sin(theta));
  }

  // Clamped coords for charge
  const safeX = clampPos(chargeX, SAFE.xMin, SAFE.xMax);
  const safeY = clampPos(chargeY, SAFE.yMin, SAFE.yMax);

  // Force vector points toward circular center (angle to center)
  const forceAngle = Math.atan2(centerY - safeY, centerX - safeX);
  const forceMagnitude = interpolate(frame, [45, 65], [0, 80], CLAMP);

  // Velocity arrow length
  const velLength = 95;
  const vEndX = safeX + Math.cos(velAngle) * velLength;
  const vEndY = safeY + Math.sin(velAngle) * velLength;

  const fEndX = safeX + Math.cos(forceAngle) * forceMagnitude;
  const fEndY = safeY + Math.sin(forceAngle) * forceMagnitude;

  // Continuous subtle pulse
  const fieldPulse = Math.sin(frame / 14) * 0.15 + 0.85;
  const glowPulse = Math.sin(frame / 8) * 4 + 18;

  // Build the traced trajectory path
  const trailPoints = useMemo(() => {
    const pts = [];
    const stepCount = 90;
    for (let i = 0; i <= stepCount; i++) {
      const f = interpolate(i, [0, stepCount], [0, Math.min(frame, 480)], CLAMP);
      if (f < 50) {
        const x = interpolate(f, [0, 50], [180, centerX], CLAMP);
        pts.push(`${x.toFixed(1)},660.0`);
      } else {
        const th = interpolate(f, [50, 480], [-Math.PI / 2, Math.PI / 2 + 0.35], CLAMP);
        const x = centerX + r * Math.cos(th);
        const y = centerY - r * Math.sin(th);
        pts.push(`${x.toFixed(1)},${y.toFixed(1)}`);
      }
    }
    return pts.length > 1 ? `M ${pts.join(" L ")}` : "";
  }, [frame]);

  const fieldIntro = spring({
    frame: Math.max(0, frame - 15),
    fps: 30,
    config: { damping: 12 },
  });

  return (
    <AbsoluteFill style={{ backgroundColor: PALETTE.base }}>
      <Backdrop />

      <svg style={SVG_FRAME}>
        <defs>
          <filter id="s4-glow" x="-30%" y="-30%" width="160%" height="160%">
            <feGaussianBlur stdDeviation="6" result="blur" />
            <feMerge>
              <feMergeNode in="blur" />
              <feMergeNode in="SourceGraphic" />
            </feMerge>
          </filter>

          <linearGradient id="s4-fieldGrad" x1="0" y1="0" x2="1" y2="1">
            <stop offset="0%" stopColor={PALETTE.support} stopOpacity={0.06 * fieldIntro} />
            <stop offset="100%" stopColor={PALETTE.supportB} stopOpacity={0.03 * fieldIntro} />
          </linearGradient>

          <marker id="s4-arrow-v" markerWidth="8" markerHeight="8" refX="6" refY="4" orient="auto">
            <path d="M 0 0 L 8 4 L 0 8 z" fill="#00E676" />
          </marker>
          <marker id="s4-arrow-f" markerWidth="8" markerHeight="8" refX="6" refY="4" orient="auto">
            <path d="M 0 0 L 8 4 L 0 8 z" fill={PALETTE.accent} />
          </marker>
        </defs>

        {/* Magnetic field boundary area */}
        <rect
          x={S4_FIELD_RECT.x}
          y={S4_FIELD_RECT.y}
          width={S4_FIELD_RECT.w}
          height={S4_FIELD_RECT.h}
          rx={20}
          fill="url(#s4-fieldGrad)"
          stroke={PALETTE.support}
          strokeWidth={2}
          strokeDasharray="8 8"
          strokeOpacity={0.4 * gridAlpha}
        />

        {/* Field badge label in top-right of field box */}
        <g opacity={gridAlpha}>
          <rect
            x={S4_FIELD_RECT.x + S4_FIELD_RECT.w - 230}
            y={S4_FIELD_RECT.y + 18}
            width={210}
            height={44}
            rx={10}
            fill="#0E1729"
            stroke={PALETTE.support}
            strokeWidth={1.5}
          />
          <text
            x={S4_FIELD_RECT.x + S4_FIELD_RECT.w - 125}
            y={S4_FIELD_RECT.y + 47}
            textAnchor="middle"
            fill={PALETTE.support}
            fontSize={20}
            fontFamily="monospace"
            fontWeight="bold"
          >
            B ⊗ (Into Page)
          </text>
        </g>

        {/* Grid of 'X' markers for B-field directed into screen */}
        {Array.from({ length: S4_GRID_ROWS }).map((_, row) => {
          const gy = S4_FIELD_RECT.y + 42 + row * 76;
          return Array.from({ length: S4_GRID_COLS }).map((__, col) => {
            const gx = S4_FIELD_RECT.x + 46 + col * 86;

                  y1={-crossSize}
                  x2={-crossSize}
                  y2={crossSize}
                  stroke={PALETTE.support}
                  strokeWidth={2.2}
                />
              </g>
            );
          });
        })}

        {/* Traced trajectory path with glowing dashed trail */}
        {trailPoints && (
          <path
            d={trailPoints}
            fill="none"
            stroke={PALETTE.accent}
            strokeWidth={3.5}
            strokeDasharray="6 6"
            opacity={0.85}
          />
        )}

        {/* Instantaneous velocity vector (Green) */}
        {frame > 10 && (
          <g>
            <line
              x1={safeX}
              y1={safeY}
              x2={vEndX}
              y2={vEndY}
              stroke="#00E676"
              strokeWidth={4}

              fontSize={24}
              fontWeight="bold"
              textAnchor="middle"
            >
              v
            </text>
          </g>
        )}

        {/* Magnetic Lorentz Force vector (Yellow, orthogonal) */}
        {frame > 50 && forceMagnitude > 10 && (
          <g>
            <line
              x1={safeX}
              y1={safeY}
              x2={fEndX}
              y2={fEndY}
              stroke={PALETTE.accent}

  );

  // Continuous propagation parameter
  const phase = (frame * 0.12) % (Math.PI * 2);

  const waveStep = 12;

            <stop offset="100%" stopColor="#0B0F1D" stopOpacity="0.95" />

            <stop offset="0%" stopColor={PALETTE.support} stopOpacity="0.9" />
            <stop offset="100%" stopColor={PALETTE.support} stopOpacity="0.1" />
          </linearGradient>

          <linearGradient id="s9-b-grad" x1="0%" y1="0%" x2="100%" y2="100%">
            <stop offset="0%" stopColor={PALETTE.supportB} stopOpacity="0.85" />
            <stop offset="100%" stopColor={PALETTE.supportB} stopOpacity="0.1" />
          </linearGradient>

          <filter id="s9-glow-e" x="-40%" y="-40%" width="180%" height="180%">
            <feGaussianBlur stdDeviation="6" result="blur" />
            <feMerge>
              <feMergeNode in="blur" />
              <feMergeNode in="SourceGraphic" />
            </feMerge>
          </filter>

          <filter id="s9-glow-b" x="-40%" y="-40%" width="180%" height="180%">
            <feGaussianBlur stdDeviation="6" result="blur" />
            <feMerge>
              <feMergeNode in="blur" />
              <feMergeNode in="SourceGraphic" />
            </feMerge>
          </filter>

          <marker id="s9-arrow-e" viewBox="0 0 10 10" refX="5" refY="5" markerWidth="4" markerHeight="4" orient="auto-start-reverse">
            <path d="M 0 1.5 L 10 5 L 0 8.5 z" fill={PALETTE.support} />
          </marker>
          <marker id="s9-arrow-b" viewBox="0 0 10 10" refX="5" refY="5" markerWidth="4" markerHeight="4" orient="auto-start-reverse">
            <path d="M 0 1.5 L 10 5 L 0 8.5 z" fill={PALETTE.supportB} />
          </marker>
        </defs>

        {/* ----------------- PHASE 1 & 2: EQUATION CARDS ----------------- */}
        {collapseProgress < 1 && (
          <g opacity={1 - collapseProgress}>
            {S9_EQUATIONS.map((eq, idx) => {
              const cardDelay = idx * 6;
              const cardSpring = spring({ frame: frame - cardDelay, fps: 30, config: { damping: 16, stiffness: 90 } });
              const cardYInitial = ZONES.stage.y + 40 + idx * 125;
              const cardXInitial = ZONES.stage.x + 80;

              // Collapse toward stage center
              const targetX = ZONES.stage.cx - 240;
              const targetY = ZONES.stage.cy - 45;
              const curX = interpolate(collapseProgress, [0, 1], [cardXInitial, targetX], CLAMP);
              const curY = interpolate(collapseProgress, [0, 1], [cardYInitial, targetY], CLAMP);
              const curScale = interpolate(collapseProgress, [0, 1], [1, 0.15], CLAMP) * cardSpring;

              const cardH = 100;

              return (
                <g key={idx} transform={`translate(${curX}, ${curY}) scale(${curScale})`}>

                  <rect
                    x={0}
                    y={0}
                    width={cardW}
                    height={cardH}
                    rx={12}
                    fill="url(#s9-card-grad)"
                    stroke={isCoupled ? PALETTE.accent : PALETTE.support}
                    strokeWidth={isCoupled ? 2 : 1.5}
                    strokeOpacity={isCoupled ? linkPulse : 0.6}
                    filter={isCoupled ? "url(#s9-glow-e)" : undefined}
                  />

                  {/* Left accent indicator bar */}
                  <rect
                    x={0}
                    y={0}
                    width={8}
                    height={cardH}
                    rx={4}
                    fill={idx % 2 === 0 ? PALETTE.support : PALETTE.supportB}
                  />

                  {/* Text rendered via SVG */}
                  <text
                    x={30}
                    y={38}
                    fill={PALETTE.accent}

                    fontWeight={700}

                    x={30}

            {/* Coupling visual links between Faraday (idx 2) & Ampère-Maxwell (idx 3) */}

                  d={`M ${ZONES.stage.x + 780} ${ZONES.stage.y + 340} Q ${ZONES.stage.x + 850} ${ZONES.stage.y + 395} ${ZONES.stage.x + 780} ${ZONES.stage.y + 450}`}

              strokeWidth={2}
              opacity={0.5}

              const sinVal = Math.sin(pt.k - phase);

              // Electric Field vector (pure vertical along Y)
              const ey = -sinVal * eAmpMax;
              // Magnetic Field vector (skewed along isometric Z plane)
              const bx = sinVal * bAmpMax;

              // Only draw vector arrows on periodic intervals to avoid clutter

              return (
                <g key={i}>
                  {/* Electric Vector Arrow */}
                  {isMajor && Math.abs(ey) > 8 && (
                    <line
                      x1={xPos}
                      y1={waveCenterY}
                      x2={xPos}

                      opacity={0.7}

                      y2={waveCenterY + by}

                      opacity={0.7}

              const ePath = activePts.reduce((acc, pt, idx) => {
                const x = waveStartX + pt.xOffset;
                const y = waveCenterY - Math.sin(pt.k - phase) * eAmpMax;

                const sinVal = Math.sin(pt.k - phase);
                const x = waveStartX + pt.xOffset + sinVal * bAmpMax;
                const y = waveCenterY + sinVal * bSkewY;
                return idx === 0 ? `M ${x} ${y}` : `${acc} L ${x} ${y}`;
              }, "");

              return (
                <g>
                  <path
                    d={ePath}
                    fill="none"
                    stroke={PALETTE.support}
                    strokeWidth={4}
                    filter="url(#s9-glow-e)"

                    d={bPath}
                    fill="none"
                    stroke={PALETTE.supportB}
                    strokeWidth={4}
                    filter="url(#s9-glow-b)"

              const xPos = waveStartX + pt.xOffset;
              const sinVal = Math.sin(pt.k - phase);

              const bx = sinVal * bAmpMax;
              const by = sinVal * bSkewY;

              <rect
                x={0}
                y={0}
